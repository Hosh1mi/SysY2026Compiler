别名分析（BasicAliasAnalysis）

BasicAliasAnalysis 分析指针之间的关系：是否指向同一内存位置（别名）， 以及函数对内存的副作用。头文件 src/include/mid/analysis/basicAliasAnalysis.hpp。

别名结果

结果含义
NoAlias两个指针一定不指向同一位置
MayAlias可能指向同一位置（保守）
MustAlias一定指向同一位置

例子：局部变量 vs 全局变量

// SysY 源码
int g;          // 全局变量
int foo() {
    int a = 1;  // 局部变量
    g = 2;
    return a;   // a 和 g 一定不别名
}
// IR
@g = global i32 0

%a = alloca i32        ; 局部栈分配
store i32 1, %a
store i32 2, ptr @g    ; 全局地址
%ret = load i32, %a

// 别名分析结果：
BasicAliasAnalysis &AA = AM.getBasicAA(module);
AA.alias(%a, @g);       // → NoAlias（不同内存对象）

例子：函数副作用

// 纯函数（无副作用）
int add(int a, int b) {
    return a + b;
}

// 有副作用的函数
void set_global(int x) {
    g = x;
}
BasicAliasAnalysis &AA = AM.getBasicAA(module);
AA.isPure(add_func);           // → true（不访问全局内存）
AA.mayHaveSideEffect(set_fn);  // → true（写全局变量）

// 查询函数对特定指针的影响
AA.getModRefInfo(call_inst, %local_ptr);  // → NoModRef（不影响局部指针）
AA.getFunctionModRef(func, @g);           // → Mod（修改全局变量 g）

例子：数组指针分析

// SysY 源码
void copy(int a[], int b[], int n) {
    for (int i = 0; i < n; i++) {
        b[i] = a[i];
    }
}
// IR
%gep_a = getelementptr i32*, i32* %a, i32 %i   ; a[i]
%gep_b = getelementptr i32*, i32* %b, i32 %i   ; b[i]
%val = load i32, %gep_a
store i32 %val, %gep_b

// 别名分析
AA.alias(%gep_a, %gep_b);  // → MayAlias（无法证明 a != b）
AA.alias(%a, %b);           // → MayAlias（参数指针，保守）
AA.isLocalArrayPointer(%a); // → false（来自函数参数）

查询接口

class BasicAliasAnalysis {
    // 别名查询
    AliasResult alias(Value *a, Value *b) const;
    AliasResult alias(const MemoryLocation &a, const MemoryLocation &b) const;

    // 指令对指针的影响
    ModRefInfo getModRefInfo(Instruction *inst, Value *ptr) const;

    // 函数级别的副作用汇总
    ModRefInfo getFunctionModRef(Function *func, Value *ptrOrGlobal = nullptr) const;
    bool isPure(Function *func) const;
    bool mayHaveSideEffect(Function *func) const;
    bool isLocalArrayPointer(Value *ptr) const;
};
